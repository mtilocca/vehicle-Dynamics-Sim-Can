cmake_minimum_required(VERSION 3.20.0)

# ── MCUboot image signing key ─────────────────────────────────────────────────
# Must be set before find_package(Zephyr) so cmake/mcuboot.cmake picks it up.
set(MCUBOOT_SIGNING_KEY_FILE ${CMAKE_CURRENT_SOURCE_DIR}/mcuboot_signing_key.pem)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(hdv_sim VERSION 1.0.0 LANGUAGES CXX C)

set(REPO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)


# Phase 0
set(APP_SRCS
    src/main.cpp
)

# Phase 1: shell, LED, HTTP server (now HTTPS)
list(APPEND APP_SRCS
    src/shell/shell_auth.cpp
    src/shell/shell_plant.cpp
    src/shell/shell_can.cpp
    src/shell/shell_stats.cpp
    src/shell/shell_vehicle.cpp
    src/shell/shell_network.cpp
    src/shell/shell_system.cpp
    src/shell/shell_mqtt.cpp
)
list(APPEND APP_SRCS src/led/led_task.cpp)
list(APPEND APP_SRCS
    src/http/http_server.cpp
    src/http/http_cmd.cpp
    src/http/http_session.cpp
    src/http/http_ota.cpp
    src/http/page/http_page_root.cpp
    src/http/page/http_page_plant.cpp
    src/http/page/http_page_actuator.cpp
    src/http/page/http_page_can.cpp
    src/http/page/http_page_controls.cpp
    src/http/page/http_page_ctrl_source.cpp
    src/http/page/http_page_mqtt.cpp
    src/http/page/http_page_resources.cpp
    src/http/page/http_page_threads.cpp
    src/http/page/http_page_memory.cpp
    src/http/page/http_page_api.cpp
    src/http/page/http_page_login.cpp
)

# Phase 3c: common TLS credentials library (HTTPS + future MQTTS)
list(APPEND APP_SRCS src/tls/tls_creds.cpp)

# Phase 3: CAN RX thread
list(APPEND APP_SRCS src/can/can_rx.cpp)

# Phase 4: Plant model, CAN TX
file(GLOB_RECURSE PLANT_SRCS "${REPO_ROOT}/src/plant/*.cpp")
list(FILTER PLANT_SRCS EXCLUDE REGEX ".*/plant_model\\.cpp$")
list(APPEND APP_SRCS
    ${PLANT_SRCS}
    ${REPO_ROOT}/src/plant/subsystem_manager/subsystem_manager.cpp
    src/plant/plant_model_zephyr.cpp
    src/plant/plant_thread.cpp
    src/can/can_tx.cpp
)

# Auxiliary threads
list(APPEND APP_SRCS
    src/stats/stats_thread.cpp
    src/watchdog/watchdog_thread.cpp
)

# MQTT client — subscribes to hdv/cmd/actuator, publishes hdv/state/vehicle
list(APPEND APP_SRCS src/mqtt/mqtt_client.cpp)

target_sources(app PRIVATE ${APP_SRCS})

# ── Embed dashboard CSS ───────────────────────────────────────────────────────
generate_inc_file_for_target(app
    ${CMAKE_CURRENT_SOURCE_DIR}/src/http/dashboard.css
    ${CMAKE_CURRENT_BINARY_DIR}/dashboard.css.inc
)

# ── Embed TLS certificate and private key ─────────────────────────────────────
# Generated by scripts/gen_tls_certs.sh — gitignored, never committed.
# Re-run gen_tls_certs.sh then rebuild to rotate the certificate.
generate_inc_file_for_target(app
    ${CMAKE_CURRENT_SOURCE_DIR}/certs/server.crt
    ${CMAKE_CURRENT_BINARY_DIR}/server_cert.inc
)
generate_inc_file_for_target(app
    ${CMAKE_CURRENT_SOURCE_DIR}/certs/server.key
    ${CMAKE_CURRENT_BINARY_DIR}/server_key.inc
)

# ── Embed MQTT broker CA certificate ─────────────────────────────────────────
# Generated by scripts/gen_mqtt_broker_cert.sh — gitignored, never committed.
# Re-run gen_mqtt_broker_cert.sh (with broker IP) then rebuild when broker IP changes.
generate_inc_file_for_target(app
    ${CMAKE_CURRENT_SOURCE_DIR}/certs/broker_ca.crt
    ${CMAKE_CURRENT_BINARY_DIR}/broker_ca.inc
)


target_include_directories(app PRIVATE
    ${REPO_ROOT}
    ${REPO_ROOT}/src
    ${REPO_ROOT}/src/plant
    ${REPO_ROOT}/utils
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/http
    ${CMAKE_CURRENT_SOURCE_DIR}/src/http/page
    ${CMAKE_CURRENT_SOURCE_DIR}/cpp_compat
    ${CMAKE_CURRENT_BINARY_DIR}    # dashboard.css.inc, server_cert.inc, server_key.inc
)
